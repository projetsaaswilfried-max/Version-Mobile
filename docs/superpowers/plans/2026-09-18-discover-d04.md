bool shouldShowDiscoverPremiumSheet({
  required String subscriptionStatus,
  bool isStaff = false,
}) {
  if (isStaff) return false;
  if (subscriptionStatus == 'ACTIVE') return false;
  if (subscriptionStatus == 'EXPIRED') return false;
  return subscriptionStatus == 'FREE' || subscriptionStatus == 'CANCELED';
}

class DiscoverPremiumPlan {
  const DiscoverPremiumPlan({required this.label, required this.priceLabel});
