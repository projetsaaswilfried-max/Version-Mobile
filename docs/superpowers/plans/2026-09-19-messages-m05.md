```dart
class MessageImageViewerPage extends StatelessWidget {
  const MessageImageViewerPage({
    super.key,
    required this.imageUrl,
    this.title,
  });

  final String imageUrl;
  final String? title;

  @override
  Widget build(BuildContext context) {
    return Theme(
      data: AppTheme.dark,
      child: Builder(
        builder: (context) {
          final colors = context.colors;
          return Scaffold(
            backgroundColor: colors.background,
            appBar: AppBar(
              backgroundColor: colors.background,
              foregroundColor: colors.textPrimary,
              elevation: 0,
              title: title == null ? null : Text(title!),
            ),
            body: Center(
              child: AppNetworkImage(
                url: imageUrl,
                fit: BoxFit.contain,
                semanticLabel: title ?? 'Image',
              ),
            ),
          );
        },
      ),
    );
  }
}
```

4. `conversation_thread_page` `onMediaTap` :

```dart
onMediaTap: (message) {
  if (message.type == MessageType.image &&
      message.mediaUrl != null &&
      message.mediaUrl!.isNotEmpty) {
    Navigator.of(context).push(
      MaterialPageRoute<void>(
        builder: (_) => MessageImageViewerPage(
          imageUrl: message.mediaUrl!,
          title: message.mediaLabel,
        ),
      ),
    );
    return;
  }
  ScaffoldMessenger.of(context).showSnackBar(
    const SnackBar(content: Text('Média : bientôt')),
  );
},
```
