```dart
// lib/features/messages/domain/composer/thread_composer_mode.dart
enum ThreadComposerMode { canCompose, inviteActions, waitingPeer, readOnlySystem }

ThreadComposerMode threadComposerMode({
  required String status,
  required bool isSystem,
  required bool isIncomingInvite,
}) {
  if (isSystem) return ThreadComposerMode.readOnlySystem;
  final s = status.trim().toUpperCase();
  if (s == 'PENDING') {
    return isIncomingInvite
        ? ThreadComposerMode.inviteActions
        : ThreadComposerMode.waitingPeer;
  }
  return ThreadComposerMode.canCompose;
}
```

```dart
// lib/features/messages/domain/entities/message.dart
import 'package:equatable/equatable.dart';

enum MessageType { text, image, video, document, voice, system }

class Message extends Equatable {
  const Message({
    required this.id,
    required this.conversationId,
    required this.type,
    required this.createdAt,
    this.content,
    this.senderIsMe = false,
    this.deleted = false,
    this.mediaLabel,
    this.fileName,
    this.durationSeconds,
  });

  final String id;
  final String conversationId;
  final MessageType type;
  final DateTime createdAt;
  final String? content;
  final bool senderIsMe;
  final bool deleted;
  final String? mediaLabel;
  final String? fileName;
  final int? durationSeconds;

  @override
  List<Object?> get props => [
        id, conversationId, type, createdAt, content, senderIsMe, deleted,
        mediaLabel, fileName, durationSeconds,
      ];
}

String messageDisplayText(Message message) {
  if (message.deleted) return 'Message supprimé';
  return message.content ?? '';
}
```
